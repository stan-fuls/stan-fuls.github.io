{%- comment -%} 文档「上一篇 / 下一篇」导航 (按 frontmatter 的 date 时间排序) - 顺序由每篇文档的 date 决定: 上一篇 = 时间更早的一篇 (更旧) 下一篇 = 时间更晚的一篇 (更新) - 模板已自动写入 date (创建时间), 新增文档推送后即自动生效, 无需改任何额外文件 - 仅当文档数 >= 2 时才显示导航 {%- endcomment -%} {%- assign docs = site.pages | where_exp: "p", "p.path contains 'docs/knowledge-docs/'" | where: "layout", "doc" -%} {%- if docs.size > 1 -%} {%- comment %} 构造可排序的键: 时间(YYYYMMDDHHMMSS) + "|" + url {% endcomment -%} {%- assign keys = "" | split: "" -%} {%- for p in docs -%} {%- assign d = p.date | default: "0000-00-00 00:00:00" -%} {%- assign ds = d | date: "%Y%m%d%H%M%S" -%} {%- assign k = ds | append: "|" | append: p.url -%} {%- assign keys = keys | push: k -%} {%- endfor -%} {%- assign keys = keys | sort -%} {%- comment %} 定位当前文章 {% endcomment -%} {%- assign cur = page.url | remove_last: '/' -%} {%- assign idx = -1 -%} {%- for k in keys -%} {%- assign parts = k | split: "|" -%} {%- assign u = parts[1] | remove_last: '/' -%} {%- if u == cur -%}{%- assign idx = forloop.index0 -%}{%- endif -%} {%- endfor -%} {%- if idx >= 0 -%}
{%- if idx > 0 -%} {%- assign pk = keys[ idx | minus: 1 ] -%} {%- assign pparts = pk | split: "|" -%} {%- assign prev = docs | where: "url", pparts[1] | first -%} ← 上一篇 {{ prev.title }} {%- else -%} ← 上一篇 没有更早的文章 {%- endif -%} {%- assign next_idx = idx | plus: 1 -%} {%- if next_idx < keys.size -%} {%- assign nk = keys[ next_idx ] -%} {%- assign nparts = nk | split: "|" -%} {%- assign next = docs | where: "url", nparts[1] | first -%} 下一篇 → {{ next.title }} {%- else -%} 下一篇 → 没有更新的文章 {%- endif -%}
{%- endif -%} {%- endif -%}
